/** Pure, default-denied grant contract. No database, UI role inference or grant side effects.
 * Witnesses must be constructed from current rows under the reviewed transaction locks.
 * These checks never replace patient/source/assessment/reference authority or session checks.
 */
export const ORTHO_CLINICAL_APPROVAL = "ortho_assessment_analysis_approval" as const;
export type GrantResult<T> = { ok: true; value: T } | { ok: false; code: "invalid_command" | "forbidden" | "identity_changed" | "stale_grant" | "grant_required" };
export interface ClinicalGrantRef { anchorId: number; version: number; eventId: number }
export interface ClinicalGrantCommand {
  commandId: string; capability: typeof ORTHO_CLINICAL_APPROVAL; action: "grant" | "revoke";
  userId: number; doctorPartyId: number; expected: ClinicalGrantRef | null; reason: string;
}
export interface CurrentClinicalUser {
  id: number; active: boolean; role: "admin" | "doctor" | "reception" | "accountant";
  doctorPartyId: number | null;
}
export interface CurrentDoctorParty { id: number; kind: string; active: boolean }
export interface CurrentClinicalGrant extends ClinicalGrantRef {
  userId: number; doctorPartyId: number; enabled: boolean;
}
export interface CurrentManagementActor { id: number; active: boolean; role: string; sessionCurrent: boolean }
const fail = (code: Extract<GrantResult<never>, { ok: false }>["code"]): GrantResult<never> => ({ ok: false, code });
const id = (value: unknown): value is number => typeof value === "number" && Number.isSafeInteger(value) && value > 0;
const record = (value: unknown): value is Record<string, unknown> => !!value && typeof value === "object" && !Array.isArray(value)
  && (Object.getPrototypeOf(value) === Object.prototype || Object.getPrototypeOf(value) === null);
const exact = (value: Record<string, unknown>, names: readonly string[]) => Object.keys(value).length === names.length && names.every(name => Object.hasOwn(value, name));
export function checkClinicalGrantRef(raw: unknown): GrantResult<ClinicalGrantRef> {
  if (!record(raw) || !exact(raw, ["anchorId", "version", "eventId"]) || !id(raw.anchorId) || !id(raw.version) || !id(raw.eventId)) return fail("invalid_command");
  return { ok: true, value: { anchorId: raw.anchorId, version: raw.version, eventId: raw.eventId } };
}
export function checkClinicalGrantCommand(raw: unknown): GrantResult<ClinicalGrantCommand> {
  if (!record(raw) || !exact(raw, ["commandId", "capability", "action", "userId", "doctorPartyId", "expected", "reason"])
    || typeof raw.commandId !== "string" || !/^[A-Za-z0-9_-]{16,80}$/.test(raw.commandId)
    || raw.capability !== ORTHO_CLINICAL_APPROVAL || (raw.action !== "grant" && raw.action !== "revoke")
    || !id(raw.userId) || !id(raw.doctorPartyId) || typeof raw.reason !== "string" || !raw.reason.trim() || raw.reason.length > 500
    || /[\u0000-\u0008\u000b\u000c\u000e-\u001f]/.test(raw.reason)) return fail("invalid_command");
  const expected = raw.expected === null ? null : checkClinicalGrantRef(raw.expected);
  if (expected && !expected.ok || raw.action === "revoke" && expected === null) return fail("invalid_command");
  return { ok: true, value: { commandId: raw.commandId, capability: ORTHO_CLINICAL_APPROVAL, action: raw.action,
    userId: raw.userId, doctorPartyId: raw.doctorPartyId, expected: expected?.ok ? expected.value : null, reason: raw.reason } };
}
const same = (a: ClinicalGrantRef, b: ClinicalGrantRef) => a.anchorId === b.anchorId && a.version === b.version && a.eventId === b.eventId;
const validGrant = (g: CurrentClinicalGrant) => id(g.anchorId) && id(g.version) && id(g.eventId) && id(g.userId) && id(g.doctorPartyId) && typeof g.enabled === "boolean";
/** Management admission only. Replay reconciliation belongs before fresh-version admission in the writer. */
export function checkClinicalGrantManagement(actor: CurrentManagementActor, command: ClinicalGrantCommand,
  target: CurrentClinicalUser | null, party: CurrentDoctorParty | null, current: CurrentClinicalGrant | null): GrantResult<ClinicalGrantCommand> {
  const checked = checkClinicalGrantCommand(command);
  if (!checked.ok) return checked;
  if (!id(actor.id) || actor.active !== true || actor.sessionCurrent !== true || actor.role !== "admin") return fail("forbidden");
  if (!target || target.id !== command.userId || !party || party.id !== command.doctorPartyId || party.kind !== "doctor") return fail("identity_changed");
  // Revocation remains possible for an inactive or relinked historical identity. It never enables anything.
  if (command.action === "grant" && (target.active !== true || (target.role !== "doctor" && target.role !== "admin")
    || target.doctorPartyId !== party.id || party.active !== true)) return fail("identity_changed");
  if (current !== null && (!validGrant(current) || current.userId !== target.id || current.doctorPartyId !== party.id)) return fail("identity_changed");
  if (command.expected === null ? current !== null : current === null || !same(command.expected, current)) return fail("stale_grant");
  return checked;
}
/** Capability admission only, after a transaction-current session witness. Missing/disabled always denies, including admin. */
export function checkClinicalApprovalGrant(user: CurrentClinicalUser | null, sessionCurrent: boolean,
  party: CurrentDoctorParty | null, current: CurrentClinicalGrant | null, expected: ClinicalGrantRef): GrantResult<CurrentClinicalGrant> {
  if (sessionCurrent !== true || !user || !id(user.id) || user.active !== true || (user.role !== "doctor" && user.role !== "admin")) return fail("forbidden");
  if (!party || !id(party.id) || party.kind !== "doctor" || party.active !== true || user.doctorPartyId !== party.id) return fail("identity_changed");
  if (!current || !validGrant(current) || current.enabled !== true) return fail("grant_required");
  if (current.userId !== user.id || current.doctorPartyId !== party.id) return fail("identity_changed");
  const requested = checkClinicalGrantRef(expected);
  if (!requested.ok || !same(requested.value, current)) return fail("stale_grant");
  return { ok: true, value: { ...current } };
}
export type AutomaticClinicalGrantRevocation = "identity_relinked" | "account_disabled" | "role_changed" | "party_disabled";
/** Returns a policy cause only; state changes and event/audit writes must be atomic with the original writer. */
export function clinicalGrantUserRevocation(before: CurrentClinicalUser, after: CurrentClinicalUser): AutomaticClinicalGrantRevocation | null {
  if (before.id !== after.id) throw new Error("Clinical grant identity changed");
  if (before.doctorPartyId !== after.doctorPartyId) return "identity_relinked";
  if (before.active === true && after.active !== true) return "account_disabled";
  if (before.role !== after.role) return "role_changed";
  return null; // Reactivation is never a grant command.
}
export function clinicalGrantPartyRevocation(before: CurrentDoctorParty, after: CurrentDoctorParty): "party_disabled" | null {
  if (before.id !== after.id || before.kind !== after.kind) throw new Error("Clinical grant party identity changed");
  return before.kind === "doctor" && before.active === true && after.active !== true ? "party_disabled" : null;
}
